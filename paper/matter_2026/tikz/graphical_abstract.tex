%% Purpose : square, single-panel summary that reads clockwise from the
%%           measured enamel to the printed lattice. Replaces the v1 image,

%% Build   : pdflatex graphical_abstract.tex
%%           pdftoppm -png -r 254 -singlefile graphical_abstract.pdf ../graphical_abstract

%% ============================================================================
